3

Define a React-free contract

packages/copilot-bridge/CopilotBridge.ts

Keep the boundary structural and imperative: plain data, schemas, and handlers enter; promises and cleanup functions return. The Host serializes context values before registration. There are no React or CopilotKit imports in this file. 

type Cleanup = () => void;

interface CopilotBridge {
  registerContext(input: {
    description: string;
    value: unknown;
    agentIds?: string[];
  }): Cleanup;

  registerTool(input: {
    name: string;
    description?: string;
    agentId?: string;
    parameters?: unknown;
    handler: (
      args: Record<string, unknown>,
      context?: {
        signal?: AbortSignal;
        agent?: unknown;
      }
    ) => Promise<unknown>;
  }): Cleanup;
}
4

Adapt CopilotKit in the Host

apps/host/src/JourneyHost.jsx

Call useCopilotKit() under the React 19 provider and memoize the adapter. It serializes context values and mirrors every add operation with its remove operation. This state-mutating POC disables automatic tool follow-up so a new agent turn cannot race the next feature context effect. Pass the stable object to both persistent feature roots. 

const { copilotkit } = useCopilotKit();
const copilotBridge = useMemo(() => ({
  registerContext(input) {
    const value = typeof input.value === "string"
      ? input.value : JSON.stringify(input.value);
    const id = copilotkit.addContext({ ...input, value });
    return () => copilotkit.removeContext(id);
  },
  registerTool(tool) {
    copilotkit.addTool({
      ...tool,
      followUp: false,
      handler: (args, context) =>
        tool.handler(args, {
          signal: context?.signal,
          agent: context?.agent
        })
    });
    return () =>
      copilotkit.removeTool(tool.name, tool.agentId);
  }
}), [copilotkit]);

<ProfileFeature
  copilotBridge={copilotBridge}
  onNavigate={navigate}
/>
<ProductFeature
  copilotBridge={copilotBridge}
  onNavigate={navigate}
/>
